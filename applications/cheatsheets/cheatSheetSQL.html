Sie ermöglicht es uns, Datenstrukturen zu definieren (DDL), Daten zu manipulieren (DML) und komplexe Abfragen zu stellen (DQL). 16.3 DQL – Data Query LanguageData Query Language

SELECT (vollständig)DQL
SELECT DISTINCT spalte1 AS alias1, COUNT(*)
FROM tabelle AS t
WHERE bedingung
GROUP BY spalte1
HAVING COUNT(*) > 5
ORDER BY spalte1 DESC
LIMIT 10 OFFSET 20;

Reihenfolge:Order: FROM → WHERE → GROUP BY → HAVING → SELECT → DISTINCT → ORDER BY → LIMIT

WHERE-OperatorenDQL
WHERE preis BETWEEN 10 AND 100
  AND kategorie IN ('A', 'B')
  AND name LIKE 'A%'
  AND rabatt IS NOT NULL
  AND NOT (ort = 'Berlin' OR ort = 'Hamburg');
JOIN-TypenDQL
SELECT * FROM a INNER JOIN b ON a.id = b.a_id;
SELECT * FROM a LEFT JOIN b ON a.id = b.a_id;
SELECT * FROM a RIGHT JOIN b ON a.id = b.a_id;
SELECT * FROM a FULL OUTER JOIN b ON a.id = b.a_id;
SELECT * FROM a CROSS JOIN b;
AggregatfunktionenDQL
SELECT
    COUNT(*) AS anzahl,
    COUNT(DISTINCT ort) AS orte,
    SUM(preis) AS summe,
    AVG(preis) AS durchschnitt,
    MIN(preis) AS min_p,
    MAX(preis) AS max_p
FROM produkte GROUP BY kategorie HAVING COUNT(*) > 5;
MengenoperatorenDQL
SELECT name FROM kunden UNION SELECT name FROM lieferanten;
SELECT name FROM kunden UNION ALL SELECT name FROM lieferanten;
SELECT name FROM kunden INTERSECT SELECT name FROM lieferanten;
SELECT name FROM kunden EXCEPT SELECT name FROM lieferanten;

SELECT name FROM kunden
WHERE umsatz > (SELECT AVG(umsatz) FROM kunden);

WHERE EXISTS (SELECT 1 FROM bestellungen b WHERE b.kunden_id = k.id);

CASE WHENDQL
SELECT name, preis,
    CASE
        WHEN preis < 10  THEN 'günstig'
        WHEN preis < 100 THEN 'mittel'
        ELSE 'teuer'
    END AS preisklasse
FROM produkte;
